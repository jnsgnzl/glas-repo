Glas-Patches, die beide Bau-Skripte gleich anwenden (Windows: windows/apply-glas.ps1, Vero: apply-glas.sh ueber
glas-patches.pl). Alle plattformfrei. Laufen nach den uebrigen Patches; Dateien, die sonst niemand patcht, setzen die
Skripte vorher auf den Kodi-Stand zurueck.

Aufbau eines Blocks (alles ausserhalb von Bloecken ist Kommentar):
  @@ after|before|replace <Datei> | <Name fuer Meldungen>
  <Anker: ganze Zeilen, muss genau so in der Datei stehen>
  @@--
  <Einsatz: ganze Zeilen; after = hinter dem Anker, before = davor, replace = statt des Ankers>
  @@end
Steht der Einsatz schon in der Datei, wird der Block uebersprungen (wiederholbar). Fehlt der Anker, bricht der Lauf ab.

Spulvorschau beschleunigt: Film fuer die Vorschau offen halten
=====================================================================================================================

Kodis Bildziehen baut fuer jedes Bild alles neu auf (Datei ueber das Netz oeffnen, Demuxer, Software-Decoder, springen,
ein Bild). Auf der Vero dauerte ein Vorschaubild so ~1,1 s (05.11.2026), Kodi sprang vorher (seekdelay 0) - die Vorschau
war nie zu sehen. GlasSeekThumb haelt Datei, Demuxer und Decoder des zuletzt gespulten Films offen und springt nur noch
(Decoder zuruecksetzen, bis zum ersten Bild dekodieren). Eine Anfrage, hinter der schon eine neuere wartet, entfaellt
(beim schnellen Weiterspulen zaehlt nur das letzte Ziel). Geschlossen wird beim Stopp der Wiedergabe (Ankuendigung
Player/OnStop), bei einem anderen Film und nach einem Fehler; Kodis eigener Weg (Kapitelbilder) bleibt unberuehrt.
Vor dem Abbau der Netzfreigaben (Ruhezustand) schliesst CApplication::CloseNetworkShares sofort, da Kodis Stopp sich
erst danach ankuendigt (sonst Zugriff auf eine abgebaute NFS-Verbindung). Braucht ein Bild ueber 1,5 s, gibt es fuer
diesen Film bis zum Stopp keine mehr.
Gemessen 05.11.2026: Vero 1080p 80-360 ms je Bild (vorher ~1,1 s), PC 55-140 ms, Blu-ray-ISO am PC 450-950 ms.

Dekodiert wird nur das Schluesselbild am Sprungziel, direkt mit libavcodec statt Kodis Decoder (05.11.2026, abends):
Kodis Software-Decoder gab das Bild erst heraus, wenn der Umsortier-Puffer voll war (HEVC: 3-5 Bilder, auf der Vero in
4K je ~1 s, zusammen 5-6 s). Jetzt: ein Paket, Decoder leeren (Drain) -> das Schluesselbild kommt sofort; ohne
Deblocking (skip_loop_filter), Slice-Threads, 10 Bit direkt beim Verkleinern nach BGRA. Der Decoder lebt nur fuer ein
Bild (wenig Speicher, auch bei 4K); Datei und Demuxer bleiben offen. Klappt das nicht (Blu-ray: nach dem Sprung kommt
nicht immer zuerst ein Schluesselbild), sucht ein zweiter Durchgang das naechste I-Bild und verwirft alles davor ungelesen
(skip_frame NONINTRA, SHOW_ALL, angenommen nur Bilder vom Typ I: Blu-rays markieren ihre I-Bilder nicht als
Schluesselbilder, NONKEY fand nichts; ohne Filter kamen graue Flaechen bzw. voll dekodiert bis dorthin 4,4 s). Der Skin
fragt waehrend eines Spulschritts im Sekundentakt nach (Player.SeekTime laeuft mit); meist ist es dasselbe

bool OpenFile(Open& o, const std::string& path)
{
  o.Close();
  CFileItem item(path, false);
  item.SetMimeTypeForInternetFile();
  o.input = CDVDFactoryInputStream::CreateInputStream(
      URIUtils::IsBluray(path) ? &g_glasNoPlayer : nullptr, item);
  if (!o.input || !o.input->Open())
  {
    o.Close();
    return false;
  }
  o.demuxer.reset(CDVDFactoryDemuxer::CreateDemuxer(o.input, true));
  if (!o.demuxer)
  {
    o.Close();
    return false;
  }
  for (CDemuxStream* stream : o.demuxer->GetStreams())
  {
    if (!stream)
      continue;
    if (stream->type == STREAM_VIDEO && !(stream->flags & AV_DISPOSITION_ATTACHED_PIC))
    {
      o.videoStream = stream->uniqueId;
      o.demuxerId = stream->demuxerId;
    }
    else
      o.demuxer->EnableStream(stream->demuxerId, stream->uniqueId, false);
  }
  CDemuxStream* video = o.videoStream != -1 ? o.demuxer->GetStream(o.demuxerId, o.videoStream) : nullptr;
  if (!video)
  {
    o.Close();
    return false;
  }
  o.hint.Assign(*video, true);
  o.decoder = avcodec_find_decoder(o.hint.codec);
  if (!o.decoder)
  {
    o.Close();
    return false;
  }
  o.path = path;
  return true;

//! Decoder fuer ein Bild: ohne Deblocking, Slice-Threads (keine Verzoegerung wie Frame-Threads)
//! intra: nur I-Bilder dekodieren (alles andere ungelesen verwerfen) und auch ohne Wiederherstellungspunkt ausgeben -
//! Blu-rays markieren ihre I-Bilder nicht als Schluesselbilder (NONKEY verwarf alles); ein I-Bild ist fuer sich ganz
AVCodecContext* NewContext(const Open& o, bool intra = false)
{
  AVCodecContext* ctx = avcodec_alloc_context3(o.decoder);
  if (!ctx)
    return nullptr;
  ctx->codec_tag = o.hint.codec_tag;
  ctx->coded_width = o.hint.width;
  ctx->coded_height = o.hint.height;
  ctx->bits_per_coded_sample = o.hint.bitsperpixel;
  ctx->bits_per_raw_sample = o.hint.bitdepth;
  if (o.hint.extradata)
  {
    ctx->extradata = static_cast<uint8_t*>(av_mallocz(o.hint.extradata.GetSize() + AV_INPUT_BUFFER_PADDING_SIZE));
    if (ctx->extradata)
    {
      ctx->extradata_size = static_cast<int>(o.hint.extradata.GetSize());
      memcpy(ctx->extradata, o.hint.extradata.GetData(), o.hint.extradata.GetSize());
    }
  }
  ctx->skip_loop_filter = AVDISCARD_ALL;
  ctx->flags2 |= AV_CODEC_FLAG2_FAST;
  if (intra)
  {
    ctx->skip_frame = AVDISCARD_NONINTRA;
    ctx->flags2 |= AV_CODEC_FLAG2_SHOW_ALL;
    // sofort ausgeben: sonst wartete das I-Bild im Umsortier-Puffer bis zum Leeren (bis zu 90 Pakete gelesen)
    ctx->flags |= AV_CODEC_FLAG_LOW_DELAY;
  }
  ctx->thread_type = FF_THREAD_SLICE;
  ctx->thread_count = 0; // so viele wie Kerne
  if (avcodec_open2(ctx, o.decoder, nullptr) < 0)
    avcodec_free_context(&ctx);
  return ctx;
}

//! Video-Pakete ab der aktuellen Stelle an den Decoder, bis ein Bild kommt; drain: danach leeren (das zuletzt
//! gegebene Bild kommt sofort, ohne auf die Umsortierung zu warten)
//! intraOnly: nur I-Bilder annehmen (Durchgang 2, SHOW_ALL gaebe sonst unvollstaendige Bilder heraus)
bool Feed(Open& o, AVCodecContext* ctx, AVFrame* frame, int maxPackets, bool drain, int& fed,
          DemuxPacket* first = nullptr, bool intraOnly = false)
{
  const auto receive = [&]() {
    while (avcodec_receive_frame(ctx, frame) == 0)
    {
      if (!intraOnly || frame->pict_type == AV_PICTURE_TYPE_I)
        return true;
      av_frame_unref(frame);
    }
    return false;
  };
  AVPacket* pkt = av_packet_alloc();
  if (!pkt)
  {
    if (first)
      CDVDDemuxUtils::FreeDemuxPacket(first);
    return false;
  }
  bool got = false;
  int reads = o.demuxer->GetNrOfStreams() * 160;
  int sent = 0;
  while (!got && sent < maxPackets && reads-- > 0)
  {
    DemuxPacket* packet = first ? first : o.demuxer->Read();
    first = nullptr;
    if (!packet)
      break;
    if (packet->iStreamId != o.videoStream || !packet->pData || packet->iSize <= 0)
    {
      CDVDDemuxUtils::FreeDemuxPacket(packet);
      continue;
    }
    pkt->data = packet->pData;
    pkt->size = packet->iSize;
    const int ret = avcodec_send_packet(ctx, pkt); // kopiert die Daten (Paket ohne eigenen Puffer)
    CDVDDemuxUtils::FreeDemuxPacket(packet);
    ++sent;
    ++fed;
    if (ret < 0 && ret != AVERROR(EAGAIN))
      continue;
    got = receive();
  }
  if (!got && drain && sent > 0)
  {
    avcodec_send_packet(ctx, nullptr);
    got = receive();
  }
  av_packet_free(&pkt);
  return got;
}

//! Kopie des zuletzt gerechneten Bildes
std::unique_ptr<CTexture> FromCache(const Open& o)
{
  std::unique_ptr<CTexture> copy = CTexture::CreateTexture(o.cacheW, o.cacheH);
  if (!copy || copy->GetPitch() != o.cachePitch || o.cachePixels.size() != o.cachePitch * o.cacheH)
    return {};
  copy->SetAlpha(false);
  copy->SetOrientation(o.cacheOrientation);
  memcpy(copy->GetPixels(), o.cachePixels.data(), o.cachePixels.size());
  return copy;
}

//! erstes Video-Paket ab der aktuellen Stelle (andere Spuren verwerfen)
DemuxPacket* FirstVideoPacket(Open& o)
{
  int reads = o.demuxer->GetNrOfStreams() * 160;
  while (reads-- > 0)
  {
    DemuxPacket* packet = o.demuxer->Read();
    if (!packet)
      return nullptr;
    if (packet->iStreamId == o.videoStream && packet->pData && packet->iSize > 0)
      return packet;
    CDVDDemuxUtils::FreeDemuxPacket(packet);
  }
  return nullptr;
}

//! springen und das Schluesselbild dort als Textur (Groesse wie ExtractThumbToTexture)
std::unique_ptr<CTexture> Decode(Open& o, int64_t ms, int& fed, bool& cached)
{
  fed = 0;
  cached = false;
  if (!o.demuxer->SeekTime(static_cast<double>(ms), true))
    return {};
  DemuxPacket* first = FirstVideoPacket(o);
  if (!first)
    return {};
  // gleiches Schluesselbild wie beim letzten Bild: Kopie, nicht neu dekodieren
  const double key = first->dts != DVD_NOPTS_VALUE ? first->dts : first->pts;
  if (key != DVD_NOPTS_VALUE && key == o.cacheKey && !o.cachePixels.empty())
  {
    CDVDDemuxUtils::FreeDemuxPacket(first);
    cached = true;
    return FromCache(o);
  }
  AVCodecContext* ctx = NewContext(o);
  AVFrame* frame = av_frame_alloc();
  if (!ctx || !frame)
  {
    CDVDDemuxUtils::FreeDemuxPacket(first);
    av_frame_free(&frame);
    avcodec_free_context(&ctx);
    return {};
  }
  // 1. das Paket am Sprungziel (Schluesselbild) und sofort leeren
  bool got = Feed(o, ctx, frame, 1, true, fed, first);
  if (!got)
  {
    // 2. neuer Decoder (nach dem Leeren nimmt er nichts mehr an), von vorn bis zum naechsten I-Bild; alles davor
    //    verwirft er ungelesen (Blu-ray: bis zu ~30 Bilder, voll dekodiert auf der Vero 4,4 s)
    avcodec_free_context(&ctx);
    ctx = NewContext(o, true);
    if (ctx && o.demuxer->SeekTime(static_cast<double>(ms), true))
      got = Feed(o, ctx, frame, 90, true, fed, nullptr, true);
  }
  std::unique_ptr<CTexture> result;
  if (got && frame->width > 0 && frame->height > 0)
  {
    double aspect = static_cast<double>(frame->width) / frame->height;
    if (frame->sample_aspect_ratio.num > 0 && frame->sample_aspect_ratio.den > 0)
      aspect *= av_q2d(frame->sample_aspect_ratio);
    if (o.hint.forced_aspect && o.hint.aspect != 0)
      aspect = o.hint.aspect;
    const unsigned int displayWidth = static_cast<unsigned int>(frame->height * aspect + 0.5);
    const unsigned int width = std::min(
        displayWidth, CServiceBroker::GetSettingsComponent()->GetAdvancedSettings()->m_imageRes);
    const unsigned int height = static_cast<unsigned int>(width / aspect);
    struct SwsContext* sws =
        width && height ? sws_getContext(frame->width, frame->height, static_cast<AVPixelFormat>(frame->format),
                                         width, height, AV_PIX_FMT_BGRA, SWS_FAST_BILINEAR, nullptr, nullptr,
                                         nullptr)
                        : nullptr;
    if (sws)
    {
      result = CTexture::CreateTexture(width, height);
      result->SetAlpha(false);
      uint8_t* dst[] = {result->GetPixels(), nullptr, nullptr, nullptr};
      int dstStride[] = {static_cast<int>(result->GetPitch()), 0, 0, 0};
      result->SetOrientation(DegreeToOrientation(o.hint.orientation));
      sws_scale(sws, frame->data, frame->linesize, 0, frame->height, dst, dstStride);
      sws_freeContext(sws);
      // fuer weitere Anfragen am selben Schluesselbild merken
      o.cacheKey = key;
      o.cacheW = width;
      o.cacheH = height;
      o.cachePitch = result->GetPitch();
      o.cacheOrientation = DegreeToOrientation(o.hint.orientation);
      o.cachePixels.assign(result->GetPixels(), result->GetPixels() + o.cachePitch * height);
    }
  }
  av_frame_free(&frame);
  avcodec_free_context(&ctx);
  return result;
}
} // namespace GLAS_SEEKTHUMB

//! Spulvorschau (VideoChapterImageFileLoader.cpp, glasseek://): Bild an der Zeit aus dem offen gehaltenen Film
std::unique_ptr<CTexture> GlasSeekThumb(const std::string& path, int seconds)
{
  using namespace GLAS_SEEKTHUMB;
  State& s = Get();
  const unsigned int ticket = ++s.latest;
  std::unique_lock<std::mutex> guard(s.lock);
  if (ticket != s.latest)
  {
    CLog::Log(LOGDEBUG, "Glas-Spulvorschau: Bild bei {} s entfaellt (neuere Anfrage)", seconds);
    return {}; // eine neuere Anfrage wartet schon: diese entfaellt
  }
  if (!s.listener)
  {
    if (const auto announcements = CServiceBroker::GetAnnouncementManager())
    {
      s.listener = new CStopListener();
      announcements->AddAnnouncer(s.listener);
    }
  }
  const unsigned int stopsNow = s.stops;
  if (!s.open.path.empty() && s.openStops != stopsNow)
    s.open.Close(); // vor einem Stopp geoeffnet
  if (!PlayerActive())
  {
    s.open.Close();
    return {};
  }
  if (s.slowPath == path && s.slowStops == stopsNow)
    return {}; // zu langsam fuer eine Vorschau: nicht nebenher weiter dekodieren
  CFileItem item(path, false);
  if (!CDVDFileInfo::CanExtract(item))
    return {};
  const auto start = std::chrono::steady_clock::now();
  if (s.open.path != path)
  {
    if (!OpenFile(s.open, path))
    {
      CLog::Log(LOGWARNING, "Glas-Spulvorschau: {} nicht zu oeffnen", CURL::GetRedacted(path));
      return {};
    }
    s.openStops = stopsNow;
    CLog::Log(LOGINFO, "Glas-Spulvorschau: Film geoeffnet ({} ms)",
              std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now() - start)
                  .count());
  }
  // Waehrend man spult, laeuft der Film weiter: der Skin fragt jede Sekunde ein neues Ziel an. Braucht ein Bild laenger
  // (Blu-ray ~1,1 s, 4K bis ~1 s), waere jedes beim Fertigwerden schon ueberholt und die Anzeige bliebe stehen; dann
  // gilt das letzte Bild fuer Anfragen bis 3 s daneben weiter, echte Spruenge (ab 10 s) rechnen sofort neu
  if (s.open.lastSlow && !s.open.cachePixels.empty() && std::abs(seconds - s.open.lastSeconds) < 3)
  {
    CLog::Log(LOGDEBUG, "Glas-Spulvorschau: Bild bei {} s wie bei {} s (letztes Bild langsam)", seconds,
              s.open.lastSeconds);
    return FromCache(s.open);
  }
  const auto decodeStart = std::chrono::steady_clock::now();
  int fed = 0;
  bool cached = false;
  auto result = Decode(s.open, static_cast<int64_t>(seconds) * 1000, fed, cached);
  if (result && !cached)
  {
    s.open.lastSeconds = seconds;
    s.open.lastSlow = std::chrono::steady_clock::now() - decodeStart > std::chrono::milliseconds(500);
  }
  const auto now = std::chrono::steady_clock::now();
  const auto took = std::chrono::duration_cast<std::chrono::milliseconds>(now - start).count();
  CLog::Log(took > 400 ? LOGINFO : LOGDEBUG, "Glas-Spulvorschau: Bild bei {} s {} ({} ms, {}, {}x{})", seconds,
            result ? "fertig" : "fehlt", took, cached ? std::string("gleiches Schluesselbild") :
            std::to_string(fed) + " Pakete", s.open.hint.width, s.open.hint.height);
  if (now - decodeStart > std::chrono::milliseconds(1500))
  {
    CLog::Log(LOGINFO, "Glas-Spulvorschau: zu langsam fuer diesen Film, aus bis zum Stopp");
    s.slowPath = path;
    s.slowStops = stopsNow;
  }
  // Datei und Demuxer bleiben offen (der Decoder lebt nur fuer ein Bild, auch bei 4K wenig Speicher)
  if (!result || s.openStops != s.stops)
    s.open.Close(); // beim naechsten Mal frisch oeffnen
  return result;
}

bool CSeekHandler::GlasConfirmAction(const CAction& action)
{
  if (!m_requireSeek || !m_glasConfirm)
    return false;

  auto& components = CServiceBroker::GetAppComponents();
  const auto appPlayer = components.GetComponent<CApplicationPlayer>();
  switch (action.GetID())
  {
    case ACTION_SELECT_ITEM:
    case ACTION_SHOW_OSD:
    case ACTION_PLAYER_PLAY:
    case ACTION_PAUSE:
    case ACTION_PLAYER_PLAYPAUSE:
    {
      {
        std::unique_lock<CCriticalSection> lock(m_critSection);
        appPlayer->SeekTimeRelative(static_cast<int64_t>(m_seekSize * 1000));
        Reset();
        m_seekChanged = true;
      }
      // wie tvOS: von der neuen Stelle aus spielen
      if (appPlayer->IsPausedPlayback())
        g_application.OnAction(CAction(ACTION_PAUSE));
      return true;
    }
    case ACTION_NAV_BACK:
    case ACTION_PREVIOUS_MENU:
    case ACTION_SHOW_GUI:
    {
      std::unique_lock<CCriticalSection> lock(m_critSection);
      Reset();
      m_seekChanged = true;
      return true;
    }
    case ACTION_STOP:
    {
      std::unique_lock<CCriticalSection> lock(m_critSection);
      Reset();
      m_seekChanged = true;
      return false; // Kodi stoppt wie gewohnt
    }
    default:
      return false;
  }
}

/*! \brief Glas: im Container (Id im aktiven Fenster bzw. Dialog) den ersten Eintrag mit dem Anfangsbuchstaben waehlen
 *  \param params (Id, Buchstabe A-Z oder #)
 */
static int JumpLetter(const std::vector<std::string>& params)
{
  if (params.size() < 2 || params[1].empty())
    return -1;
  const int id = atoi(params[0].c_str());
  auto& windows = CServiceBroker::GetGUI()->GetWindowManager();
  CGUIWindow* window = windows.GetWindow(windows.GetActiveWindowOrDialog());
  CGUIControl* control = window ? window->GetControl(id) : nullptr;
  if (!control || !control->IsContainer())
  {
    window = windows.GetWindow(windows.GetActiveWindow());
    control = window ? window->GetControl(id) : nullptr;
  }
  if (!control || !control->IsContainer())
    return -1;
  const IGUIContainer* container = dynamic_cast<const IGUIContainer*>(control);
  if (!container)
    return -1;
  std::wstring wanted;
  g_charsetConverter.utf8ToW(params[1], wanted);
  if (wanted.empty())
    return -1;
  const bool other = wanted[0] == L'#';
  const wchar_t letter = static_cast<wchar_t>(std::towupper(wanted[0]));
  const bool ignoreArticles = CServiceBroker::GetSettingsComponent()->GetSettings()->GetBool(
      CSettings::SETTING_FILELISTS_IGNORETHEWHENSORTING);
  for (int i = 0;; ++i)
  {
    const std::shared_ptr<CGUIListItem> item = container->GetListItem(i, INFOFLAG_LISTITEM_ABSOLUTE);
    if (!item)
      break;
    if (item->IsFileItem() && static_cast<const CFileItem*>(item.get())->IsParentFolder())
      continue;
    std::string label = item->GetLabel();
    if (ignoreArticles)
      label = SortUtils::RemoveArticles(label);
    std::wstring wlabel;
    g_charsetConverter.utf8ToW(label, wlabel);
    if (wlabel.empty())
      continue;
    const wchar_t first = static_cast<wchar_t>(std::towupper(wlabel[0]));
    const bool isLetter = first >= L'A' && first <= L'Z';
    if (other ? !isLetter : first == letter)
    {
      CGUIMessage msg(GUI_MSG_ITEM_SELECT, window->GetID(), id, i);
      control->OnMessage(msg);
      return 0;
    }
  }
  return 0;
}
